```ts
export function updateNode(
  tx: Tx,
  id: string,
  patch: {
    x?: number
    y?: number
    w?: number
    h?: number
    groupId?: string | null
    data?: unknown
    updatedAt: number
  },
): void {
  const set: Partial<typeof node.$inferInsert> = { updatedAt: patch.updatedAt }
  if (patch.x !== undefined) set.x = patch.x
  if (patch.y !== undefined) set.y = patch.y
  if (patch.w !== undefined) set.w = patch.w
  if (patch.h !== undefined) set.h = patch.h
  if (patch.groupId !== undefined) set.groupId = patch.groupId
  if (patch.data !== undefined) set.dataJson = JSON.stringify(patch.data)
  tx.update(node).set(set).where(eq(node.id, id)).run()
}

export function moveNodes(tx: Tx, canvasId: string, moves: Array<{ nodeId: string; x: number; y: number }>): number {
  const ids = new Set<string>()
  for (const move of moves) {
    if (ids.has(move.nodeId)) throw new Error('duplicate nodeId')
    if (!Number.isFinite(move.x) || !Number.isFinite(move.y)) throw new Error('invalid coordinate')
    ids.add(move.nodeId)
  }
  const rows = tx.select({ id: node.id }).from(node).where(eq(node.canvasId, canvasId)).all()
  const known = new Set(rows.map((row) => row.id))
  if ([...ids].some((id) => !known.has(id))) throw new Error('node not in canvas')
  for (const move of moves) {
    tx.update(node).set({ x: move.x, y: move.y, updatedAt: Date.now() }).where(eq(node.id, move.nodeId)).run()
  }
  return moves.length
}

export function createGroup(tx: Tx, canvasId: string, group: Node, childIds: string[]): void {
  if (group.type !== 'group') throw new Error('not a group')
  const children = tx.select().from(node).where(eq(node.canvasId, canvasId)).all()
  const byId = new Map(children.map((row) => [row.id, row]))
  for (const id of childIds) {
    const child = byId.get(id)
    if (!child) throw new Error('child not in canvas')
    if (child.type === 'group') throw new Error('nested group is not supported')
    if (child.groupId !== null) throw new Error('child already grouped')
  }
  insertNode(tx, group)
  for (const id of childIds) {
    tx.update(node).set({ groupId: group.id, updatedAt: group.updatedAt }).where(eq(node.id, id)).run()
  }
}

export function ungroupChildren(tx: Tx, groupId: string): number {
  const rows = tx.select({ id: node.id }).from(node).where(eq(node.groupId, groupId)).all()
  tx.update(node).set({ groupId: null }).where(eq(node.groupId, groupId)).run()
  return rows.length
}

export function setReviewPreliminary(
  tx: Tx,
  reviewId: string,
  input: { verdict: 'pass' | 'fail' | 'unknown'; reason: string; at: number },
): boolean {
  const existing = tx.select({ id: review.id }).from(review).where(eq(review.id, reviewId)).get()
  if (!existing) return false
  tx.update(review)
    .set({
      preliminaryVerdict: input.verdict,
      preliminaryReason: input.reason.slice(0, 2000),
      preliminaryAt: input.at,
    })
    .where(eq(review.id, reviewId))
    .run()
  return true
}
```

```ts
const IMAGE_MIME: Record<string, string> = {
  '.png': 'image/png',
  '.jpg': 'image/jpeg',
  '.jpeg': 'image/jpeg',
  '.webp': 'image/webp',
  '.gif': 'image/gif',
  '.svg': 'image/svg+xml',
}

app.get('/api/workspaces/:id/uploads/:name', async (req, reply) => {
  const { id, name } = req.params as { id: string; name: string }
  const workspace = getWorkspace(deps.db, id)
  if (!workspace) throw appError('workspace_not_found', 'workspace not found')
  if (name !== basename(name) || name.includes('\\')) throw appError('image_not_found', 'image not found')
  const workspacePath = resolveWorkspaceDir(deps.config.workspacesRoot, workspace.relPath)
  const uploadDir = resolveWithin(workspacePath, UPLOAD_DIR)
  let realFile: string
  try {
    const candidate = resolveWithin(uploadDir, name)
    realFile = realpathSync(candidate)
    if (!realFile.startsWith(realpathSync(uploadDir) + sep)) throw new Error('escape')
    if (lstatSync(realFile).isSymbolicLink()) throw new Error('symlink')
  } catch {
    throw appError('image_not_found', 'image not found')
  }
  const mime = IMAGE_MIME[extname(realFile).toLowerCase()]
  if (!mime) throw appError('image_type_unsupported', 'unsupported image type')
  const stat = statSync(realFile)
  if (stat.size > MAX_UPLOAD_BYTES) throw appError('image_too_large', 'image too large')
  reply.header('X-Content-Type-Options', 'nosniff')
  reply.header('Cache-Control', 'private, max-age=300')
  if (mime === 'image/svg+xml') {
    reply.header('Content-Security-Policy', "default-src 'none'; style-src 'unsafe-inline'; sandbox")
  }
  return reply.type(mime).send(createReadStream(realFile))
})
```
