# M0 骨架实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use `subagent-driven-development` (recommended) or `executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付一个可安装、可启动、可打包、可容器化、有 CI 闸门的空壳：三栏界面 + 健康检查 + WS 鉴权 + SQLite 迁移。

**Architecture:** 单仓 pnpm workspace。`packages/protocol` 提供 zod 契约；`apps/daemon` 是 Fastify 单端口服务（API + WS + 静态资源），同时是唯一发布包；`apps/web` 是 Vite SPA。M0 不接真实 Agent，只留适配器接口与空实现。

**Tech Stack:** Node 22、TypeScript ESM、Fastify 5、`@fastify/websocket`、better-sqlite3、Drizzle、React 19、Vite、`@xyflow/react`、vitest、Playwright、Docker、GitHub Actions。

**计划位置说明**：本仓库约定 `docs/plans/`，不使用工具默认的 `docs/agent-harness/plans/`。`scripts/validate-handoff.sh` 在本仓库不存在，落盘后用等价检查替代（frontmatter 字段、任务数、占位符扫描）。

**Commit 策略**：默认每个任务结束提交一次（Conventional Commits）。若希望手动统一提交，执行时告知即可。

`apps/daemon/src/config.ts`：

```ts
import { join } from 'node:path'

export interface Config {
  host: string
  port: number
  token: string | null
  dataDir: string
  webRoot: string
}

export function loadConfig(argv: string[], env: NodeJS.ProcessEnv): Config {
  let host = '127.0.0.1'
  let port = 7317
  for (let i = 0; i < argv.length; i += 1) {
    if (argv[i] === '--host') host = argv[i + 1] ?? host
    if (argv[i] === '--port') port = Number(argv[i + 1] ?? port)
  }
  const token = env.BATON_TOKEN ?? null
  if (host !== '127.0.0.1' && !token) {
    throw new Error('BATON_TOKEN is required when binding a non-loopback host')
  }
  return {
    host,
    port,
    token,
    dataDir: env.BATON_DATA_DIR ?? `${env.HOME ?? '.'}/.baton`,
    webRoot: env.BATON_WEB_ROOT ?? join(process.cwd(), 'apps/web/dist'),
  }
}
```

```ts
import { mkdtempSync, rmSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { afterEach, describe, expect, it } from 'vitest'
import { openDb, migrateDb } from '../src/db/index.js'

const dirs: string[] = []
afterEach(() => dirs.splice(0).forEach((d) => rmSync(d, { recursive: true, force: true })))

function tables(db: ReturnType<typeof openDb>): string[] {
  return db.prepare("select name from sqlite_master where type='table' order by name").all()
    .map((r) => (r as { name: string }).name)
}

describe('migrateDb', () => {
  it('is idempotent on a fresh directory', () => {
    const dir = mkdtempSync(join(tmpdir(), 'baton-db-')); dirs.push(dir)
    const file = join(dir, 'baton.db')
    const db1 = openDb(file); migrateDb(db1)
    const before = tables(db1); db1.close()
    const db2 = openDb(file); migrateDb(db2)
    expect(tables(db2)).toEqual(before)
    expect(before).toContain('task')
    db2.close()
  })
})
```

`src/db/index.ts`：

```ts
import Database from 'better-sqlite3'
import { drizzle } from 'drizzle-orm/better-sqlite3'
import { migrate } from 'drizzle-orm/better-sqlite3/migrator'
import { fileURLToPath } from 'node:url'

export function openDb(file: string) {
  const db = new Database(file)
  db.pragma('journal_mode = WAL')
  db.pragma('foreign_keys = ON')
  return db
}

export function migrateDb(db: Database.Database): void {
  const migrationsFolder = fileURLToPath(new URL('../../migrations', import.meta.url))
  migrate(drizzle(db), { migrationsFolder })
}
```

`src/ws/server.ts`：

```ts
import websocket from '@fastify/websocket'
import type { FastifyInstance } from 'fastify'
import type { Config } from '../config.js'

export async function attachWs(app: FastifyInstance, config: Config): Promise<void> {
  await app.register(websocket)
  app.get('/ws', { websocket: true }, (socket, req) => {
    const url = new URL(req.url, 'http://localhost')
    const supplied = url.searchParams.get('token')
      ?? req.headers.authorization?.replace(/^Bearer\s+/i, '')
      ?? null
    const timer = setTimeout(() => socket.close(4401, 'unauthorized'), 5000)
    if (config.token && supplied !== config.token) return
    clearTimeout(timer)
    socket.on('message', (raw) => {
      const msg = JSON.parse(String(raw))
      if (msg.type === 'ping') {
        socket.send(JSON.stringify({ v: 1, type: 'pong', id: msg.id, ts: Date.now(), payload: {} }))
      }
    })
  })
}
```

```ts
import type { FastifyInstance } from 'fastify'

export async function registerAuth(app: FastifyInstance, token: string | null) {
  app.get('/auth', async (req, reply) => {
    const supplied = (req.query as { token?: string }).token
    if (!token || supplied !== token) return reply.code(401).send({ code: 'unauthorized' })
    return reply.setCookie('baton_token', token, { httpOnly: true, sameSite: 'strict', path: '/' })

  })
}
```

```ts
export async function registerStatic(app: FastifyInstance, webRoot: string) {
  await app.register(fastifyStatic, { root: webRoot, wildcard: false })
  app.setNotFoundHandler((req, reply) => {
    if (req.url.startsWith('/api') || req.url.startsWith('/ws')) {
```
