# M6-1 Plan 3：Web 画布与任务现场

> **For agentic workers:** REQUIRED SUB-SKILL: Use agent-harness:subagent-driven-development (recommended) or agent-harness:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [x]`) syntax for tracking.

**Goal:** 让画布能渲染 image/group/review 三类节点，并让“任务现场”成为可读、可退出的派生视图。

**Architecture:** 所有派生数据在 `model.ts` 计算；组件只消费 flow node data；现场状态放 Zustand，不写数据库；分组坐标在 web 内用 parentId 视觉化成父子，持久化时转回绝对坐标。

**Tech Stack:** React 19、`@xyflow/react` 13、Zustand、TanStack Query、Tailwind 4、HeroUI v3。

**Commit strategy:** 每个 task 末尾提交；若你要最后统一提交，执行前告诉我。

**Design sync:** 所有颜色/间距/圆角使用 `DESIGN.md` token（`bg-surface`、`border-separator`、`text-muted`、`bg-accent`、`bg-warning`、`bg-danger`、`bg-success`）；新增状态必须有文字/图标，不允许只靠颜色。Task 5 的现场条和 Task 2 的 review 卡是设计验收点。

export interface CanvasNodeData extends Record<string, unknown> {
  node: ProtocolNode
  task: TaskSummary | null
  dimmed?: boolean
  imageUrl?: string
  review?: ReviewView | null
  childCount?: number
}

export interface CanvasRenderOptions {
  scene?: TaskScene | null
  reviewDetails?: Record<string, ReviewView>
  workspaceId?: string
}

const REVIEW_CARD_ID = (taskId: string) => `review:${taskId}`

export function toFlowNodes(
  nodes: ProtocolNode[],
  tasks: TaskSummary[],
  options: CanvasRenderOptions = {},
): FlowNode<CanvasNodeData>[] {
  const scene = options.scene ?? null
  const related = new Set(scene?.relatedNodeIds ?? [])
  const relatedGroups = new Set(scene?.relatedGroupIds ?? [])
  const expand = new Set(scene?.expandGroupIds ?? [])
  const groupById = new Map(nodes.filter((node) => node.type === 'group').map((node) => [node.id, node]))
  const taskByNode = /* existing latest-task map */

  const base = nodes
    .filter((node) => node.type !== 'group')
    .map((node) => {
      const group = node.groupId ? groupById.get(node.groupId) : undefined
      const hidden = Boolean(group?.type === 'group' && group.data.collapsed && !expand.has(group.id))
      return {
        id: node.id,
        type: node.type,
        position: group ? { x: node.x - group.x, y: node.y - group.y } : { x: node.x, y: node.y },
        ...(group ? { parentId: group.id, extent: 'parent' as const } : {}),
        hidden,
        style: { width: node.w, height: node.h },
        data: {
          node,
          task: taskByNode.get(node.id) ?? null,
          dimmed: scene ? !related.has(node.id) : false,
          ...(node.type === 'image' ? { imageUrl: resolveImageUrl(node.data.src, options.workspaceId) } : {}),
        },
      }
    })

  const groups = nodes
    .filter((node) => node.type === 'group')
    .map((node) => ({
      id: node.id,
      type: 'group',
      position: { x: node.x, y: node.y },
      style: { width: node.w, height: node.h },
      data: {
        node,
        task: null,
        dimmed: scene ? !relatedGroups.has(node.id) : false,
        childCount: nodes.filter((child) => child.groupId === node.id).length,
      },
    }))

  const reviewCards = tasks
    .filter((task) => task.status === 'review')
    .map((task) => {
      const agent = nodes.find((node) => node.id === task.nodeId)
      return {
        id: REVIEW_CARD_ID(task.id),
        type: 'review',
        position: agent ? { x: agent.x + agent.w + 40, y: agent.y } : { x: 0, y: 0 },
        draggable: false,
        selectable: true,
        style: { width: 280, height: 180 },
        data: {
          node: {
            id: REVIEW_CARD_ID(task.id), canvasId: agent?.canvasId ?? '', type: 'review',
            x: agent ? agent.x + agent.w + 40 : 0, y: agent?.y ?? 0,
            w: 280, h: 180, groupId: null, data: { taskId: task.id }, createdAt: 0, updatedAt: 0,
          },
          task,
          review: options.reviewDetails?.[task.id] ?? null,
        },
      }
    })

  return [...groups, ...base, ...reviewCards]
}
```

function absolutePosition(node: FlowNode<CanvasNodeData>, all: FlowNode<CanvasNodeData>[]) {
  const parent = node.parentId ? all.find((item) => item.id === node.parentId) : undefined
  return parent
    ? { x: parent.position.x + node.position.x, y: parent.position.y + node.position.y }
    : node.position
}

async function onDragStop(node: FlowNode<CanvasNodeData>) {
  if (!canvas) return
  if (node.data.node.type === 'group') {
    const children = nodes.filter((item) => item.parentId === node.id)
    const moves = [
      { nodeId: node.id, x: node.position.x, y: node.position.y },
      ...children.map((child) => ({ nodeId: child.id, ...absolutePosition(child, nodes) })),
    ]
    try {
      await send({ type: 'node.move_many', payload: { canvasId: canvas.canvas.id, moves } })
    } catch (cause) {
      toast.danger(cause instanceof Error ? cause.message : '分组移动失败')
    } finally {
      await refreshCanvas()
    }
    return
  }
  const absolute = absolutePosition(node, nodes)
  const groups = (canvas.nodes ?? []).filter((item) => item.type === 'group')
  const group = findContainingGroup({ ...absolute, w: node.data.node.w, h: node.data.node.h }, groups)
  await updateNode(node.id, { ...absolute, groupId: group?.id ?? null })
}
```
