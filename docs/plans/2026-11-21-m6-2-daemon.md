```ts
import { canInjectContext, cumulativeInjectedChars, assembleContextInjection, buildContextUpdateBlock } from '@baton/core'
import type { AppError, ContextInjection, Task, TaskTransitionEvent } from '@baton/protocol'
import type { Config } from '../config.js'
import type { Db } from '../db/types.js'
import type { BroadcastEvent, EventStore } from '../db/event-store.js'
import {
  appendTaskContextInjection,
  appendTaskPromptRaw,
  getCanvasByWorkspace,
  getTask,
  getWorkspace,
  insertAudit,
  listEdges,
  listNodes,
} from '../db/repositories.js'
import { appError } from '../errors.js'
import { buildContextContent } from '../workspace/context.js'
import { resolveWorkspaceDir } from '../workspace/resolver.js'
import type { TransitionOutcome } from './types.js'

const MAX_CUMULATIVE_INJECTION_CHARS = 96_000
const MAX_PROMPT_BLOCK_CHARS = 96_000
const DEFAULT_NOTE = '补充以下上下文'

export interface ContextServiceDeps {
  db: Db
  config: Config
  eventStore: EventStore
  applyTransition(taskId: string, event: TaskTransitionEvent): TransitionOutcome
  hasSession(taskId: string): boolean
  broadcast(event: BroadcastEvent): void
  now(): number
  newId(): string
}

export interface InjectContextInput {
  taskId: string
  nodeIds: string[]
  note?: string
  source: 'user' | 'supervisor'
  actor: string
}

export type InjectContextResult =
  | { ok: true; taskId: string; seq: number; truncated: boolean; dropped: number }
  | { ok: false; error: AppError }

export function createContextService(deps: ContextServiceDeps) {
  function prepare(
    input: InjectContextInput,
    options: { allowReview?: boolean } = {},
  ):
    | { ok: true; task: Task; injection: ContextInjection; block: string }
    | { ok: false; error: AppError } {
    const task = getTask(deps.db, input.taskId)
    if (!task) return { ok: false, error: appError('task_not_found', 'task not found', { taskId: input.taskId }) }
    const injectable = canInjectContext(task)
    const reviewAllowed = options.allowReview === true && task.status === 'review'
    if (!injectable.ok && !(reviewAllowed && injectable.reason === 'use_review_reject')) {
      return {
        ok: false,
        error: appError('task_not_injectable', 'task is not injectable', {
          taskId: task.id,
          status: task.status,
          reason: injectable.reason,
        }),
      }
    }
    if (task.status === 'review' && !reviewAllowed) {
      return { ok: false, error: appError('task_not_injectable', 'use review reject', { taskId: task.id, reason: 'use_review_reject' }) }
    }
    const workspace = getWorkspace(deps.db, task.workspaceId)
    if (!workspace) return { ok: false, error: appError('workspace_not_found', 'workspace not found', { workspaceId: task.workspaceId }) }
    const canvas = getCanvasByWorkspace(deps.db, workspace.id)
    if (!canvas) return { ok: false, error: appError('canvas_not_found', 'canvas not found', { workspaceId: workspace.id }) }
    const nodes = listNodes(deps.db, canvas.id)
    const byId = new Map(nodes.map((node) => [node.id, node]))
    const missing = input.nodeIds.filter((id) => !byId.has(id))
    if (missing.length > 0) return { ok: false, error: appError('node_not_found', 'context node not found', { nodeIds: missing }) }
    const invalid = input.nodeIds.filter((id) => {
      const node = byId.get(id)
      return node?.type === 'agent' || node?.type === 'review'
    })
    if (invalid.length > 0) return { ok: false, error: appError('node_type_not_injectable', 'node type is not injectable', { nodeIds: invalid }) }
    const workspacePath = resolveWorkspaceDir(deps.config.workspacesRoot, workspace.relPath)
    const { contentByNodeId } = buildContextContent({ nodes, workspacePath })
    const assembled = assembleContextInjection({
      targetNodeId: task.nodeId,
      selectedNodeIds: input.nodeIds,
      nodes,
      edges: listEdges(deps.db, canvas.id),
      contentByNodeId,
    })
    if (assembled.included.length === 0) return { ok: false, error: appError('context_empty', 'no injectable content') }
    const note = input.note?.trim() ? input.note.trim() : DEFAULT_NOTE
    const requestedChars = assembled.chars + note.length
    const used = cumulativeInjectedChars(task.context)
    if (used + requestedChars > MAX_CUMULATIVE_INJECTION_CHARS) {
      return { ok: false, error: appError('context_budget_exceeded', 'cumulative injection budget exceeded', { limit: MAX_CUMULATIVE_INJECTION_CHARS, used, requested: requestedChars }) }
    }
    const seq = (task.context?.injections.length ?? 0) + 1
    const injection: ContextInjection = {
      seq,
      at: deps.now(),
      source: input.source,
      note,
      nodeIds: input.nodeIds,
      included: assembled.included,
      dropped: assembled.dropped,
      truncated: assembled.truncated,
      relations: assembled.relations,
      relationsTruncated: assembled.relationsTruncated,
      chars: requestedChars,
    }
    const block = buildContextUpdateBlock({ seq, at: injection.at, source: injection.source, note, assembled })
    return { ok: true, task, injection, block }
  }

  function injectContext(input: InjectContextInput): InjectContextResult {
    const prepared = prepare(input)
    if (!prepared.ok) return prepared
    const { task, injection, block } = prepared
    if (task.status === 'waiting_input') {
      if (!deps.hasSession(task.id)) {
        return { ok: false, error: appError('session_not_active', 'no active session for task', { taskId: task.id }) }
      }
      const outcome = deps.applyTransition(task.id, { type: 'user_input', text: block, contextInjection: injection })
      if (!outcome.ok) return outcome
      return { ok: true, taskId: task.id, seq: injection.seq, truncated: injection.truncated, dropped: injection.dropped.length }
    }
    const event = deps.db.transaction((tx) => {
      appendTaskPromptRaw(tx, task.id, block, MAX_PROMPT_BLOCK_CHARS, deps.now())
      appendTaskContextInjection(tx, task.id, injection)
      insertAudit(tx, {
        id: deps.newId(),
        ts: deps.now(),
        actor: input.source === 'supervisor' ? 'supervisor' : 'human',
        action: 'task.context.inject',
        detail: { taskId: task.id, seq: injection.seq, source: input.source, nodeIds: input.nodeIds, chars: injection.chars, dropped: injection.dropped.length, truncated: injection.truncated },
      })
      if (input.source === 'supervisor') {
        insertAudit(tx, {
          id: deps.newId(),
          ts: deps.now(),
          actor: 'supervisor',
          action: 'supervisor.context.inject',
          detail: { taskId: task.id, seq: injection.seq, nodeIds: input.nodeIds, chars: injection.chars, dropped: injection.dropped.length },
        })
      }
      return deps.eventStore.appendContextInjected(tx, { taskId: task.id, seq: injection.seq, at: injection.at, source: input.source, nodeIds: input.nodeIds, chars: injection.chars, ts: deps.now() })
    })
    deps.broadcast(event)
    return { ok: true, taskId: task.id, seq: injection.seq, truncated: injection.truncated, dropped: injection.dropped.length }
  }

  return { injectContext, prepareContextInjection: prepare }
}

export type ContextService = ReturnType<typeof createContextService>
```

```ts
function decideReview(input: ReviewDecideInput): TransitionOutcome {
  const review = getReview(deps.db, input.reviewId)
  if (!review) {
    return {
      ok: false,
      error: appError('review_not_found', 'review not found', { reviewId: input.reviewId }),
    }
  }
  if (review.verdict !== null) {
    return {
      ok: false,
      error: appError('already_decided', 'review already decided', {
        kind: 'review',
        reviewId: input.reviewId,
      }),
    }
  }
  const current = getReviewByTask(deps.db, review.taskId)
  if (!current || current.id !== review.id) {
    return {
      ok: false,
      error: appError('review_expired', 'review is superseded by a newer attempt', {
        reviewId: input.reviewId,
      }),
    }
  }

  if (input.decision === 'approve') {
    const outcome = applyTransition(review.taskId, { type: 'approve', reviewId: input.reviewId })
    if (outcome.ok && (input.reason || review.verifyExitCode !== 0 || review.verifyExitCode === VERIFY_SKIPPED)) {
      deps.db.transaction((tx) => {
        insertAudit(tx, {
          id: newId(),
          ts: now(),
          actor: 'human',
          action: 'review.approve_override',
          detail: {
            taskId: review.taskId,
            reviewId: review.id,
            verifyExitCode: review.verifyExitCode,
            reasonProvided: Boolean(input.reason?.trim()),
          },
        })
      })
    }
    return outcome
  }

  let contextInjection: ContextInjection | undefined
  let contextBlock: string | undefined
  let feedback = input.feedback?.trim() ?? ''
  if (input.contextNodeIds?.length) {
    const prepared = contextService.prepareContextInjection(
      {
        taskId: review.taskId,
        nodeIds: input.contextNodeIds,
        ...(input.contextNote ? { note: input.contextNote } : {}),
        source: 'user',
        actor: 'user',
      },
      { allowReview: true },
    )
    if (!prepared.ok) return prepared
    contextInjection = prepared.injection
    contextBlock = prepared.block
  }
  if (!feedback) feedback = '请结合补充的上下文修改'
  return applyTransition(review.taskId, {
    type: 'reject',
    reviewId: input.reviewId,
    feedback,
    ...(contextInjection ? { contextInjection, contextBlock } : {}),
  })
}
```

注意：`prepareContextInjection` 对 `review` 状态会拒绝（`canInjectContext` 返回 use_review_reject）。review reject 需要一个允许 review 的 prepare 分支；在 context service 的 `prepare` 加参数 `allowReview = false`，review 路径传 `true`，只跳过 `canInjectContext` 的 review 拒绝，仍校验节点/预算。
