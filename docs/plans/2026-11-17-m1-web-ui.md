```ts
import {
  CanvasResponse,
  TaskDetailResponse,
  TaskListResponse,
  WorkspaceCreateInput,
  WorkspaceCreateResponse,
  WorkspaceListResponse,
} from '@baton/protocol'

async function request(path: string, init?: RequestInit): Promise<unknown> {
  const response = await fetch(path, { credentials: 'same-origin', ...init })
  if (!response.ok) {
    const body = (await response.json().catch(() => null)) as {
      code?: string
      message?: string
    } | null
    throw new Error(body?.message ?? `${response.status} ${response.statusText}`)
  }
  return response.json()
}

export async function listWorkspaces() {
  return WorkspaceListResponse.parse(await request('/api/workspaces')).workspaces
}

export async function createWorkspace(input: WorkspaceCreateInput) {
  return WorkspaceCreateResponse.parse(
    await request('/api/workspaces', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify(input),
    }),
  )
}

export async function listTasks(workspaceId: string) {
  return TaskListResponse.parse(
    await request(`/api/tasks?workspaceId=${encodeURIComponent(workspaceId)}`),
  ).tasks
}

export async function getCanvas(workspaceId: string) {
  return CanvasResponse.parse(
    await request(`/api/canvas?workspaceId=${encodeURIComponent(workspaceId)}`),
  )
}

export async function getTask(taskId: string) {
  return TaskDetailResponse.parse(await request(`/api/tasks/${encodeURIComponent(taskId)}`))
}
```

```ts
export function readUrlState() {
  const params = new URLSearchParams(window.location.search)
  return { workspaceId: params.get('workspace'), taskId: params.get('task') }
}

export function writeUrlState(next: { workspaceId?: string | null; taskId?: string | null }) {
  const params = new URLSearchParams(window.location.search)
  for (const [key, value] of Object.entries(next)) {
    if (value) params.set(key, value)
    else params.delete(key)
  }
  const query = params.toString()
  window.history.replaceState(null, '', query ? `?${query}` : window.location.pathname)
}
```

```ts
import type { ServerMessage } from '@baton/protocol'
import type { WsStatus } from '../stores/app'

interface WsClientOptions {
  onStatus: (status: WsStatus) => void
  onEvent: (event: ServerMessage) => void
}

export function createWsClient(options: WsClientOptions) {
  let socket: WebSocket | null = null
  let closed = false
  let retry = 0
  let timer: number | undefined
  let lastSeq = 0

  const sendResume = () => {
    if (socket?.readyState !== WebSocket.OPEN) return
    socket.send(
      JSON.stringify({
        v: 1,
        id: crypto.randomUUID(),
        type: 'resume',
        ts: Date.now(),
        payload: { sinceSeq: lastSeq },
      }),
    )
  }

  const schedule = () => {
    retry += 1
    const delay = Math.min(5000, 1000 * 2 ** (retry - 1))
    timer = window.setTimeout(connect, delay)
  }

  const connect = () => {
    options.onStatus(retry === 0 ? 'connecting' : 'reconnecting')
    socket = new WebSocket(
      `${location.protocol === 'https:' ? 'wss' : 'ws'}://${location.host}/ws`,
    )
    socket.addEventListener('open', () => {
      retry = 0
      options.onStatus('online')
      sendResume()
    })
    socket.addEventListener('message', (message) => {
      try {
        const parsed = JSON.parse(String(message.data)) as {
          seq?: number
          type?: string
          payload?: unknown
        }
        if (typeof parsed.seq === 'number') lastSeq = parsed.seq
        if (parsed.type && parsed.type !== 'ack' && parsed.type !== 'pong') {
          options.onEvent(parsed as ServerMessage)
        }
      } catch {
        // 非法消息不影响后续事件；daemon 仍会继续推送。
      }
    })
    socket.addEventListener('close', () => {
      if (closed) return
      options.onStatus('offline')
      schedule()
    })
    socket.addEventListener('error', () => socket?.close())
  }

  connect()

  return {
    close: () => {
      closed = true
      if (timer) window.clearTimeout(timer)
      socket?.close()
    },
    lastSeq: () => lastSeq,
  }
}
```

```tsx
import { useQuery, useQueryClient } from '@tanstack/react-query'
import { useEffect } from 'react'
import { ConnectionStatus } from './features/shared/ConnectionStatus'
import { CreateWorkspaceDialog } from './features/workspaces/CreateWorkspaceDialog'
import { WorkspaceSwitcher } from './features/workspaces/WorkspaceSwitcher'
import { listWorkspaces } from './lib/api'
import { queryKeys } from './lib/query'
import { readUrlState, writeUrlState } from './lib/url'
import { createWsClient } from './lib/ws'
import { useAppStore } from './stores/app'

export function App() {
  const activeWorkspaceId = useAppStore((state) => state.activeWorkspaceId)
  const setWorkspace = useAppStore((state) => state.setWorkspace)
  const setWsStatus = useAppStore((state) => state.setWsStatus)
  const queryClient = useQueryClient()
  const workspaces = useQuery({ queryKey: queryKeys.workspaces, queryFn: listWorkspaces })

  useEffect(() => {
    if (activeWorkspaceId) return
    const url = readUrlState()
    if (url.workspaceId) {
      setWorkspace(url.workspaceId)
    } else if (workspaces.data?.[0]) {
      setWorkspace(workspaces.data[0].id)
    }
  }, [activeWorkspaceId, setWorkspace, workspaces.data])

  useEffect(() => {
    if (activeWorkspaceId) writeUrlState({ workspaceId: activeWorkspaceId })
  }, [activeWorkspaceId])

  useEffect(() => {
    if (!activeWorkspaceId) {
      setWsStatus('offline')
      return
    }
    const client = createWsClient({
      onStatus: setWsStatus,
      onEvent: () => {
        void queryClient.invalidateQueries({ queryKey: queryKeys.tasks(activeWorkspaceId) })
        void queryClient.invalidateQueries({ queryKey: ['task'] })
      },
    })
    return () => client.close()
  }, [activeWorkspaceId, queryClient, setWsStatus])

  return (
```

```ts
import type { TaskStatus, TaskSummary } from '@baton/protocol'

export type BoardColumnId = 'unplanned' | 'todo' | 'running' | 'waiting' | 'review' | 'done' | 'closed'

export const BOARD_COLUMNS: Array<{ id: BoardColumnId; label: string }> = [
  { id: 'unplanned', label: '未规划' },
  { id: 'todo', label: '待办' },
  { id: 'running', label: '进行中' },
  { id: 'waiting', label: '待审核' },
  { id: 'review', label: '待验收' },
  { id: 'done', label: '已完成' },
  { id: 'closed', label: '已取消' },
]

export function statusColumn(status: TaskStatus): BoardColumnId {
  switch (status) {
    case 'todo':
    case 'queued':
      return 'todo'
    case 'running':
    case 'verifying':
      return 'running'
    case 'waiting_input':
      return 'waiting'
    case 'review':
      return 'review'
    case 'done':
      return 'done'
    case 'cancelled':
    case 'failed':
      return 'closed'
  }
}

export function groupTasks(tasks: TaskSummary[]): Record<BoardColumnId, TaskSummary[]> {
  const grouped: Record<BoardColumnId, TaskSummary[]> = {
    unplanned: [],
    todo: [],
    running: [],
    waiting: [],
    review: [],
    done: [],
    closed: [],
  }
  for (const task of tasks) {
    grouped[statusColumn(task.status)].push(task)
  }
  return grouped
}
```

```tsx
import { useEffect, useState } from 'react'

export function formatElapsed(since: number, now = Date.now()): string {
  const minutes = Math.max(0, Math.floor((now - since) / 60_000))
  if (minutes < 1) return '刚刚'
  if (minutes < 60) return `已等 ${minutes} 分钟`
  const hours = Math.floor(minutes / 60)
  return `已等 ${hours} 小时 ${minutes % 60} 分`
}

export function WaitingTime({ since }: { since: number }) {
  const [now, setNow] = useState(() => Date.now())
  useEffect(() => {
    const timer = window.setInterval(() => setNow(Date.now()), 30_000)
    return () => window.clearInterval(timer)
  }, [])
  return <span className="text-xs text-warning">{formatElapsed(since, now)}</span>
}
```

```tsx
import { useQuery } from '@tanstack/react-query'
import { getTask } from '../../lib/api'
import { queryKeys } from '../../lib/query'
import { useAppStore } from '../../stores/app'
import { AttentionBadge } from './AttentionBadge'
import { StatusChip } from './StatusChip'

export function TaskDetailPanel() {
  const activeWorkspaceId = useAppStore((state) => state.activeWorkspaceId)
  const activeTaskId = useAppStore((state) => state.activeTaskId)
  const detail = useQuery({
    queryKey: queryKeys.task(activeTaskId ?? ''),
    queryFn: () => getTask(activeTaskId!),
    enabled: Boolean(activeTaskId),
  })

  return (
    <aside
      data-testid="task-detail"
      className="w-[380px] shrink-0 overflow-y-auto border-l border-separator bg-surface p-4"
    >
      {!activeWorkspaceId ? (
        <p className="text-sm text-muted">尚未选择工作区</p>
      ) : !activeTaskId ? (
        <p className="text-sm text-muted">选择一个任务</p>
      ) : detail.isLoading ? (
        <p className="text-sm text-muted">加载中…</p>
      ) : detail.isError ? (
        <p className="text-sm text-danger">任务加载失败</p>
      ) : detail.data ? (
        <div className="flex flex-col gap-3">
          <div className="flex items-start justify-between gap-2">
            <h2 className="text-base font-medium">{detail.data.task.title}</h2>
            <AttentionBadge attention={detail.data.attention} />
          </div>
          <div className="flex items-center gap-2 text-xs text-muted">
            <StatusChip status={detail.data.task.status} />
            <span>{detail.data.task.engine}</span>
          </div>
          <p className="text-sm text-muted">详情、transcript 与人闸门将在 Plan 3 接入。</p>
        </div>
      ) : null}
    </aside>
  )
}
```
